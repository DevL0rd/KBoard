add_subdirectory(ui)
add_subdirectory(settings)

foreach(module typing sound emoji gif clipboard voice gamepad ui settings)
    get_property(module_tests DIRECTORY ${module} PROPERTY TESTS)
    if(module_tests)
        set_tests_properties(${module_tests} DIRECTORY ${module} PROPERTIES LABELS "unit;${module}")
    endif()
endforeach()
